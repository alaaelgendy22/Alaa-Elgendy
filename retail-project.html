# ── Feature Engineering: RFM Scores & Customer Lifetime Value ──

import pandas as pd
import numpy  as np
from datetime import datetime

def compute_rfm(df: pd.DataFrame, snapshot_date: datetime) -> pd.DataFrame:
    """
    Computes Recency, Frequency, and Monetary (RFM) scores
    for customer segmentation and churn prediction features.
    """
    rfm = df.groupby('customer_id').agg(
        Recency   = ('order_date',  lambda x: (snapshot_date - x.max()).days),
        Frequency = ('order_id',    'count'),
        Monetary  = ('revenue',     'sum')
    ).reset_index()

    # Quintile scoring: 5 = best, 1 = worst
    for col in ['Frequency', 'Monetary']:
        rfm[f'{col}_score'] = pd.qcut(rfm[col], q=5, labels=[1,2,3,4,5])

    rfm['Recency_score'] = pd.qcut(rfm['Recency'], q=5, labels=[5,4,3,2,1])
    rfm['RFM_Score']    = rfm[['Recency_score', 'Frequency_score', 'Monetary_score']].astype(int).sum(axis=1)

    # Customer Lifetime Value (simple predictive CLV)
    rfm['CLV'] = rfm['Monetary'] / rfm['Frequency'].clip(lower=1) * np.log1p(rfm['Frequency'])

    return rfm
